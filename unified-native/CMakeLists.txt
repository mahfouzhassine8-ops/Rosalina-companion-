cmake_minimum_required(VERSION 3.22)
project(rosalina_unified_workers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ROSALINA_VULKAN "Compile isolated Vulkan candidates" OFF)
if(NOT DEFINED SD_SOURCE)
  message(FATAL_ERROR "SD_SOURCE must point to the pinned stable-diffusion.cpp source")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare.py ${CMAKE_CURRENT_BINARY_DIR}/generated RESULT_VARIABLE PREPARE_RESULT)
if(NOT PREPARE_RESULT EQUAL 0)
  message(FATAL_ERROR "Preservation patch preparation failed")
endif()
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_VULKAN ${ROSALINA_VULKAN} CACHE BOOL "" FORCE)
set(GGML_VULKAN ${ROSALINA_VULKAN} CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory(${SD_SOURCE} sd-build)
foreach(ENGINE image motion)
  add_executable(rosalina-${ENGINE} ${CMAKE_CURRENT_BINARY_DIR}/generated/${ENGINE}.cpp)
  target_include_directories(rosalina-${ENGINE} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${SD_SOURCE}/ggml/include)
  target_link_libraries(rosalina-${ENGINE} PRIVATE stable-diffusion)
  if(ROSALINA_VULKAN)
    target_compile_definitions(rosalina-${ENGINE} PRIVATE ROSALINA_VULKAN=1)
  endif()
  if(ANDROID)
    # User scope correction: do not link the legacy app-lifetime guard.
    # Explicit Stop/cancellation/reap remain owned by the common worker session.
    target_link_options(rosalina-${ENGINE} PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
  endif()
endforeach()
